def safe_usage(response):
    u = response.usage
    return {"input_tokens": u.input_tokens, "output_tokens": u.output_tokens,
            "cached_input_tokens": u.input_tokens_details.cached_tokens,
            "cache_write_tokens": getattr(u.input_tokens_details, "cache_write_tokens", 0),
            "total_tokens": u.total_tokens}


def safe_error(exc, stage):
    return {"stage": stage, "error_type": type(exc).__name__,
            "cause_type": type(exc.__cause__).__name__ if exc.__cause__ else None,
            "frames": [{"file": Path(f.filename).name, "function": f.name, "line": f.lineno}
                       for f in traceback.extract_tb(exc.__traceback__)]}


class ObservedModel(Model):
    def __init__(self, wrapped, record):
        self.wrapped, self.record = wrapped, record

    async def get_response(self, *args, **kwargs):
        self.record["stage"] = "model_request"
        self.record["requests_started"] += 1
        response = await self.wrapped.get_response(*args, **kwargs)
        self.record["usage_observations"].append(safe_usage(response))
        self.record["stage"] = "runner_processing"
        return response

    async def stream_response(self, *args, **kwargs):
        raise NotImplementedError("non-streaming study")
        yield


async def episode(case, arm, artifact, store, client):
    record = {"case_id": case["case_id"], "arm": arm, "stage": "prefetch",
              "requests_started": 0, "usage_observations": [], "continuation_calls": []}
    started = time.perf_counter()
    try:
        context, backend = old.initial_context(case, arm, artifact, store)
        record.update(context["metadata"], model_input_digest=old.object_digest(context["model_input"]))
        model = ObservedModel(OpenAIProvider(openai_client=client).get_model(old.MODEL), record)
        agent = Agent(name="comment-window-recovery", instructions=old.PROMPT, model=model,
            model_settings=old.settings(), tools=old.recovery_tools(backend, case, record["continuation_calls"]),
            output_type=old.Answer)
        result = await asyncio.wait_for(Runner.run(agent, context["model_input"], max_turns=old.MAX_TURNS,
            run_config=RunConfig(tracing_disabled=True)), timeout=old.TIMEOUT)
        record["stage"] = "answer_validation"
        record["answer"] = old.Answer.model_validate(result.final_output).model_dump()
        record["correct"] = record["answer"] == old.offline.source_answer(case, store)
        record["stage"] = "usage_validation"
        usage, metrics = old.usage_metrics(result.raw_responses, old.price_table())
        record.update(status="completed", usage=usage, metrics=metrics)
    except Exception as exc:
        record.update(status="failed", error=safe_error(exc, record["stage"]))
    record["elapsed_ms"] = (time.perf_counter() - started) * 1000
    return record


async def main():
    from dotenv import load_dotenv
    from openai import AsyncOpenAI
    root = old.ROOT / "experiments/live_recovery_completion"
    root.mkdir(parents=True, exist_ok=True)
    out = root / "diagnostic.json"
    if out.exists():
        raise ValueError("diagnostic already exists; no replay")
    frozen, store, artifact = old.verify(old.OUT)
    prior = json.loads((old.OUT / "checkpoint.json").read_text())
    charge = sum(r.get("metrics", {}).get("estimated_cost_usd", 0) for r in prior["rows"]) + prior["unknown_cost_reserve_usd"]
    case = next(c for c in frozen["cases"] if c["case_id"] == "6273:ordinary:5")
    state = {"purpose": "failure diagnosis only; excluded from primary results", "aggregate_approved_cap_usd": 200,
        "prior_estimate_and_reserve_usd": charge, "script_sha256": old.digest(Path(__file__)),
        "prior_checkpoint_sha256": old.digest(old.OUT / "checkpoint.json"),
        "planned_repetitions": 3, "rows": [], "pending": None}
    load_dotenv(Path('/Users/rezarahimi/Documents/GitHub/personal/guarded-agentic-compaction/.env'))
    async with AsyncOpenAI(max_retries=0) as client:
        for i in range(3):
            if charge + 1 > 200:
                break
            state["pending"] = i
            old.write_json(out, state)
            row = await episode(case, "full", artifact, store, client)
            charge += row.get("metrics", {}).get("estimated_cost_usd", 0) + (1 if row["status"] != "completed" else 0)
            state["rows"].append(row)
            state["pending"] = None
            state["aggregate_estimate_and_reserve_usd"] = charge
            old.write_json(out, state)
            print(json.dumps(row), flush=True)
            if row["status"] == "failed":
                break


if __name__ == '__main__':
    asyncio.run(main())
