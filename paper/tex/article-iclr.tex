% Fonts. The ICLR template calls \usepackage{times}. Under pdfTeX that resolves
% to Nimbus Roman, but this paper is built with Tectonic (XeTeX), where the
% legacy `times' package does NOT resolve: the body silently falls back to Latin
% Modern and every \textbf/\emph/\textit/\textsc renders as upright regular
% text. newtxtext/newtxmath are the maintained Times-compatible replacements and
% resolve under both engines; the embedded body face is TeX Gyre Termes, i.e.
% Times, as the template requires. \openbox and \Bbbk are released because
% newtxmath and amssymb/amsthm both declare them.

\providecommand{\keywords}[1]{}
% In one column a starred float has no meaning, but LaTeX still routes
% figure*/table* through \@dblfloat, which may only be placed at the top of a
% *later* page. That strands half-empty pages. Map them onto ordinary floats so
% they take part in normal placement while the shared body keeps the starred
% form for the two-column build.
\makeatletter
\let\@dblfloat\@float
\let\end@dblfloat\end@float
\makeatother

% The ICLR style loads natbib in author-year mode. Every one of body.tex's 66
% citations is written as "...text~\cite{key}", i.e. parenthetically, which
% natbib's author-year \cite would render as "...text~Kim et al. (2024)".
% \citep gives the intended "...text~(Kim et al., 2024)". Deferred to
% \AtBeginDocument so it lands after hyperref and cleveref have wrapped \cite.
\AtBeginDocument{\let\cite\citep}
