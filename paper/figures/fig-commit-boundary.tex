  okl/.style={lane, draw=gacline!55, fill=gacgrey},

% ------------------------------------------------------ shared pre-emission band
\node[chk] (band) at (0,0)
  {\textbf{Exact for both paths --- every check that can reject \emph{before} emission}\\
   lifecycle $\cdot$ signature $\cdot$ manifest pins $\cdot$ hard guard $H$ $\cdot$
   already-observed tools $\cdot$ quota attestation $\cdot$ calibrated gate $q\le\eta$};

\node[tally] (tal) at (104,0)
  {\textbf{Terminal edges of \cref{alg:dispatch}}\\
   1 compacted execution $\cdot$ every clean rejection returns the baseline $\cdot$
   an unattested abort, failed commit, or post-commit failure raises an incident.};

% -------------------------------------------------------------- the commit line
\draw[commit] (94,-16) -- (94,-62);
\node[ghd, text=gacred!85, anchor=south] at (94,-15.4) {commit line};

% ---------------------------------------------------------------- lane A: runner
\node[ghd, text=gacteal!65!black] at (0,-18.4) {A\quad outer runner \normalfont\itshape --- owns the entry snapshot and the commit boundary};
\node[stg] (a1) at (0,-23)  {staged execution\\nothing model-visible has been written};
\node[okl] (a2) at (98,-23) {\textbf{clean fallback is exact}\\the baseline agent sees
                             byte-identical model-visible input at every failure point};
\draw[flow] (a1.east) -- (a2.west);

% --------------------------------------------------------------- lane B: adapter
\node[ghd, text=gacred!85] at (0,-41.4) {B\quad model-boundary adapter \normalfont\itshape --- the host runner owns session history};
\node[stg] (b1) at (0,-46)  {staged execution\\then a synthesized \code{ModelResponse} is
                             returned};
\node[badl] (b2) at (98,-46) {\textbf{weaker post-emission semantics}\\the host may already
                              have committed the item; the \code{Model} interface cannot
                              withdraw it, so a later verifier failure delegates with
                              history the baseline would never have contained};
\draw[flow] (b1.east) -- (b2.west);

\node[gnote, text width=92mm] at (0,-64)
  {Read every ``unmodified baseline'' claim in this paper as scoped to a staging-owning
   runner.  The adapter is guarded substitution, not byte-identical restoration, and the
   difference is structural rather than an implementation gap.};

\end{tikzpicture}
