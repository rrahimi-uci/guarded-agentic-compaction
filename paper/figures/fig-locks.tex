% Four independent claims, tested at three different times. Sized for one column
% (~80mm) so it can sit next to the prose it explains in both builds rather than
% floating to the top of a page as a spanning float.
% Authored artwork (TikZ), included by both the ACM and the article build.
% Colours are declared in each wrapper preamble so the figure is class-neutral.
\begin{tikzpicture}[
  x=1mm, y=1mm,
  font=\sffamily\scriptsize,
  lock/.style={
    draw=gacline!85, line width=0.5pt, rounded corners=1.4pt, fill=white,
    anchor=north west, align=left, inner sep=2.2pt,
    minimum width=78mm, text width=73mm, font=\sffamily\tiny},
  compile/.style={lock, draw=gacblue!85, fill=gacblue!7},
  entry/.style={lock, draw=gacteal!85, fill=gacteal!9},
  after/.style={lock, draw=orange!80!black, fill=orange!10},
  phase/.style={font=\sffamily\bfseries\tiny, text=gacblue, anchor=north west},
  gnote/.style={font=\sffamily\tiny, text=gacline, anchor=north west, align=left},
  arr/.style={-{Latex[length=1.4mm,width=1.1mm]}, draw=gacline!70, line width=0.45pt},
  norescue/.style={draw=gacred!80, line width=0.5pt, densely dashed,
                   -{Latex[length=1.4mm,width=1.1mm]}},
]

% ------------------------------------------------------------- compile time
\node[phase] at (0,4.4) {compile time \normalfont\itshape --- before $P$ exists};
\node[compile] (l1) at (0,0)
  {\textbf{Lock 1 \;$\mathcal{L}$ \;provenance}\\
   \emph{Can every argument slot be constructed?}  Eq.~\eqref{eq:grounding} and
   \cref{alg:patg}.  A slot with no witness, or with more witnesses than the ambiguity
   cap, ends the region here.};

% ------------------------------------------------------------- dispatch time
\node[phase] at (0,-15.6) {dispatch time \normalfont\itshape --- before anything is staged};
\node[entry] (l2) at (0,-20)
  {\textbf{Lock 2 \;$H$ \;hard guard}\\
   \emph{Is this entry state eligible?}  Types, required fields, categorical sets,
   numeric intervals, patterns, isolation keys, allowed effects, manifest pins.};
\node[entry] (l3) at (0,-38)
  {\textbf{Lock 3 \;$q(z)\le\eta$ \;risk gate}\\
   \emph{Is dispatch admitted under the sealed certificate?}  Eq.~\eqref{eq:cp}.  This
   is a frequency-level licence, not a probability that this one answer is right.};

% ---------------------------------------------------------- after execution
\node[phase] at (0,-53.6) {after staged execution \normalfont\itshape --- before release};
\node[after] (l4) at (0,-58)
  {\textbf{Lock 4 \;$V$ \;verifier}\\
   \emph{Do the produced receipts conform?}  Live-out presence, type, cardinality,
   empirical hulls, provenance, effect multisets, call counts --- likely invariants, not
   proofs for all future inputs.};

\foreach \a/\b in {l1/l2, l2/l3, l3/l4} {\draw[arr] (\a.south west) ++(3,0) -- ++(0,-2);}

\node[gnote, text=gacred!90, text width=78mm] at (0,-76)
  {\textbf{One-way.}  A later lock can reject what an earlier one passed; no later lock
   can open a door an earlier one closed.  ``Constructible,'' ``entry eligible,''
   ``risk admitted,'' and ``conformant after execution'' are four different claims, and
   no amount of calibration evidence substitutes for the first.};

\end{tikzpicture}
