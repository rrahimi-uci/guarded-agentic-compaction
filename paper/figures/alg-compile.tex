% Algorithm 1 (presentation order: the compiler cascade).
\begin{algorithm}[t]
\caption{The end-to-end pipeline an operator runs, from qualified traces to a registered
artifact.  Every \textbf{retire} is recorded with its stage so rejections stay in the
denominator.  Stages marked \textsc{(caller)} are separate library entry points rather than
steps inside \code{compile\_grc()}, which receives the split object and does not invoke the
feasibility estimator itself; the cascade is presented in one place because that is the
order in which the stages must run, not because one function performs all of them.}
\label{alg:compile}
\small
\begin{algorithmic}[1]
\Require episodes $\mathcal{E}$; effect catalog $\mathcal{C}$; entry schema $\Theta$;
         risk budget $\alpha$; confidence budget $\delta$; grid $\Lambda$
\Ensure artifact $A=(P,H,V,q,\eta,M)$, or \retire{} with an attributed reason
\State $\mathcal{E}\gets\textsc{Qualify}(\mathcal{E},\mathcal{C})$
       \Comment{drop partial runs, unpinned manifests, missing payloads}
\State $\{\mathcal{E}_M\}\gets\textsc{PartitionBy}(\mathcal{E},\text{manifest},
       \text{principal},\text{isolation})$
\ForAll{partitions $\mathcal{E}_M$}
  \State $(\mathcal{T},\mathcal{D},\mathcal{K},\mathcal{S})\gets
         \textsc{GroupedSplit}(\mathcal{E}_M)$ \hfill\textsc{(caller)}
         \Comment{train / dev / calibration / sealed test, split by \emph{group}}
  \State $G\gets\textsc{BuildPatg}(\mathcal{T},\Theta)$
         \Comment{typed provenance search}
  \State $\mathcal{W}\gets\textsc{MineRegions}(G,\mathcal{C},w_{\max})$
  \State $\mathcal{F}\gets\textsc{Canonicalize}(\mathcal{W})$
         \Comment{hash signature, topology, run and live-in/out shape}
  \State \textbf{keep} $F\in\mathcal{F}$ with cross-group support (and cross-day, if configured)
  \If{$\varphi\,k / n_B < \Delta$} \hfill\textsc{(caller)}
    \State \Return \retire{} \textbf{(infeasible ceiling)}
           \Comment{a separate estimator pass; run it before paying for synthesis}
  \EndIf
  \ForAll{families $F$ ranked by score \eqref{eq:score}}
    \State $P\gets\textsc{Synthesize}(F,\mathcal{L})$
           \Comment{closed library, depth $\le 2$, group-wise refit}
    \If{$P=\bot$} \State \textbf{continue} \textbf{(ungroundable slot)} \EndIf
    \State $(H,V)\gets\textsc{InduceContract}(F)$
    \If{$\neg\textsc{ReplayAndChallenge}(P,V,\mathcal{D})$}
           \Comment{requires a sandbox; records \code{perturbations\_claimed} otherwise}
      \State \textbf{continue} \textbf{(counterexample)}
             \Comment{stale reads, reordering, empty sets, drift, faults}
    \EndIf
    \State $q\gets\textsc{FitScore}(\mathcal{D})$; \textbf{freeze} $q$
           \Comment{dev groups only; never calibration}
    \State $\eta\gets\textsc{Calibrate}(q,\mathcal{K},\Lambda,\alpha,\delta)$
           \Comment{fixed-grid exact admission}
    \If{$\eta=\bot$} \State \textbf{continue} \textbf{(no admissible threshold)} \EndIf
    \State \Return $\textsc{Package}(P,H,V,q,\eta,M)$ with evidence and lifecycle
  \EndFor
\EndFor
\State \Return \retire{}
\end{algorithmic}
\end{algorithm}
